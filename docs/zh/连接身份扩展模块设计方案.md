## 详细模块设计

### 1. extensions/identity_provider/types.py - 数据结构

```python
from dataclasses import dataclass, field
from typing import Any

@dataclass
class IdentityInfo:
    """身份信息数据结构"""
    user_id: str | None = None
    domain_id: str | None = None
    app_id: str | None = None
    extra: dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> dict[str, Any]:
        """序列化为字典，仅包含非 None 字段"""
        result = {}
        if self.user_id is not None:
            result["user_id"] = self.user_id
        if self.domain_id is not None:
            result["domain_id"] = self.domain_id
        if self.app_id is not None:
            result["app_id"] = self.app_id
        if self.extra:
            result["extra"] = self.extra
        return result
```

### 2. extensions/identity_provider/base.py - 抽象基类

```python
from abc import ABC, abstractmethod

class IdentityProviderBase(ABC):
    """身份信息提供者抽象基类
    
    业务层需继承此类并实现 fetch_identity() 方法。
    在 jiuwenclaw 启动前通过 IdentityStore.register_provider() 注册。
    """
    
    @abstractmethod
    async def fetch_identity(self) -> IdentityInfo:
        """获取身份信息
        
        业务层实现此方法，返回 User ID、Domain ID、Application ID。
        实现方式由业务层决定：
        - 调用外部 API
        - 读取配置文件
        - 从环境变量获取
        - 查询数据库
        
        Returns:
            IdentityInfo: 身份信息对象，字段可为 None
        """
        ...
    
    async def on_fetch_failed(self, error: Exception) -> IdentityInfo | None:
        """获取失败时的回调（可选实现）
        
        默认返回 None（允许连接继续）。
        业务层可覆盖此方法实现自定义失败处理或返回 fallback 身份。
        
        Args:
            error: 获取失败时的异常
            
        Returns:
            IdentityInfo | None: 返回 None 表示允许连接继续
        """
        return None
```

### 3. extensions/identity_provider/store.py - 全局存储单例

```python
import logging

class IdentityStore:
    """身份信息全局存储（单例模式）
    
    存储 AgentServer 获取的身份信息，供日志系统读取。
    """
    
    _instance: IdentityStore | None = None
    
    def __init__(self) -> None:
        self._identity: IdentityInfo | None = None
        self._provider: IdentityProviderBase | None = None
        self._fetched: bool = False
    
    @classmethod
    def get_instance(cls) -> IdentityStore:
        """获取单例实例"""
        if cls._instance is None:
            cls._instance = cls()
        return cls._instance
    
    @classmethod
    def reset_instance(cls) -> None:
        """重置单例（用于测试）"""
        cls._instance = None
    
    def register_provider(self, provider: IdentityProviderBase) -> None:
        """注册身份提供者"""
        self._provider = provider
    
    def unregister_provider(self) -> None:
        """注销身份提供者"""
        self._provider = None
        self._identity = None
        self._fetched = False
    
    def get_identity(self) -> IdentityInfo | None:
        """获取当前身份信息（供日志系统读取）"""
        return self._identity
    
    def is_fetched(self) -> bool:
        """是否已获取身份信息"""
        return self._fetched
    
    async def fetch_and_store(self) -> IdentityInfo | None:
        """调用 provider 获取身份并存储
        
        获取失败时调用 on_fetch_failed 回调，
        若返回 fallback 身份则存储，否则继续连接。
        """
        if self._provider is None:
            self._fetched = True
            return None
        
        try:
            identity = await self._provider.fetch_identity()
            self._identity = identity
            self._fetched = True
            return identity
        except Exception as e:
            self._fetched = True
            fallback = await self._provider.on_fetch_failed(e)
            if fallback is not None:
                self._identity = fallback
            return self._identity
```

### 4. utils.py 修改 - 日志组件

#### IdentityFieldFilter - 日志过滤器

```python
class IdentityFieldFilter(logging.Filter):
    """身份信息字段过滤器
    
    自动为每条日志添加 user_id、domain_id、app_id 字段。
    从 IdentityStore 单例读取当前身份信息。
    """
    
    def filter(self, record: logging.LogRecord) -> bool:
        identity = IdentityStore.get_instance().get_identity()
        if identity is not None:
            record.user_id = identity.user_id
            record.domain_id = identity.domain_id
            record.app_id = identity.app_id
        else:
            record.user_id = None
            record.domain_id = None
            record.app_id = None
        return True
```

#### IdentityTextFormatter - 文本格式 Formatter

```python
class IdentityTextFormatter(logging.Formatter):
    """支持身份字段的文本格式 Formatter
    
    在文本日志中添加身份字段，仅在有值时输出。
    格式为：user_id=xxx domain_id=xxx app_id=xxx
    """
    
    def format(self, record: logging.LogRecord) -> str:
        # 构建身份字段字符串
        identity_parts = []
        if record.user_id is not None:
            identity_parts.append(f"user_id={record.user_id}")
        if record.domain_id is not None:
            identity_parts.append(f"domain_id={record.domain_id}")
        if record.app_id is not None:
            identity_parts.append(f"app_id={record.app_id}")
        
        # 供格式字符串使用 %(identity_str)s
        record.identity_str = " " + " ".join(identity_parts) + " " if identity_parts else ""
        return super().format(record)
```

#### JsonUserVisibleFormatter 扩展

在 `add_fields()` 方法中添加身份字段处理：

```python
# 添加身份字段（user_id、domain_id、app_id）
user_id = getattr(record, 'user_id', None)
domain_id = getattr(record, 'domain_id', None)
app_id = getattr(record, 'app_id', None)
# 仅添加非 None 的字段，保持 JSON 清洁
if user_id is not None:
    ordered_record['user_id'] = user_id
if domain_id is not None:
    ordered_record['domain_id'] = domain_id
if app_id is not None:
    ordered_record['app_id'] = app_id

# 在添加其他字段时，过滤掉 null 值的身份字段
identity_keys = {'user_id', 'domain_id', 'app_id'}
for key, value in log_record.items():
    if key not in ordered_record:
        if key in identity_keys and value is None:
            continue  # 跳过 null 身份字段
        ordered_record[key] = value
```

### 5. agent_ws_server.py 修改 - 触发时机

```python
async def _connection_handler(self, ws: Any) -> None:
    """处理单个 Gateway WebSocket 连接"""
    # ... 初始化代码 ...
    
    # 触发身份获取（在发送 connection.ack 之前）
    try:
        from jiuwenclaw.extensions.identity_provider import IdentityStore
        identity = await IdentityStore.get_instance().fetch_and_store()
        if identity is not None:
            logger.info(
                "[AgentWebSocketServer] 身份信息已获取: user_id=%s domain_id=%s app_id=%s",
                identity.user_id, identity.domain_id, identity.app_id,
            )
    except Exception as e:
        logger.warning("[AgentWebSocketServer] 身份获取异常: %s", e)
    
    # 发送 connection.ack
    ack_frame = {"type": "event", "event": "connection.ack", "payload": {"status": "ready"}}
    await ws.send(json.dumps(ack_frame, ensure_ascii=False))
    
    # ... 后续处理 ...
```

## 业务层集成示例

```python
# 业务层代码（不修改 jiuwenclaw）

from jiuwenclaw.extensions.identity_provider import (
    IdentityProviderBase,
    IdentityInfo,
    IdentityStore,
)
import httpx

class MyIdentityProvider(IdentityProviderBase):
    """业务层身份提供者实现"""
    
    def __init__(self, api_url: str):
        self._api_url = api_url
        self._client = httpx.AsyncClient()
    
    async def fetch_identity(self) -> IdentityInfo:
        """调用内部 API 获取身份信息"""
        resp = await self._client.get(self._api_url, timeout=5.0)
        data = resp.json()
        return IdentityInfo(
            user_id=data.get("user_id"),
            domain_id=data.get("domain_id"),
            app_id=data.get("app_id"),
        )
    
    async def on_fetch_failed(self, error: Exception) -> IdentityInfo | None:
        """失败时返回默认值或 None"""
        return IdentityInfo()  # 字段全为 None


# 在启动 jiuwenclaw 前注册
provider = MyIdentityProvider("https://internal-api.example.com/identity")
IdentityStore.get_instance().register_provider(provider)

# 启动 AgentServer
# 当 Gateway 连接时，会自动调用 provider.fetch_identity()
```

## 观测指标 Labels

身份信息获取成功后，`IdentityStore` 中保存的 `user_id`、`domain_id`、`app_id`
也会作为 OpenTelemetry metric attributes 注入到所有 20 个现有指标中。

指标侧使用 telemetry 命名空间，输出字段为：

| IdentityInfo 字段 | Metric Label |
|---|---|
| `user_id` | `jiuwenclaw.user.id` |
| `domain_id` | `jiuwenclaw.domain.id` |
| `app_id` | `jiuwenclaw.app.id` |

注入位置位于 `jiuwenclaw/telemetry/metrics.py` 的统一指标 Label 处理逻辑。
Counter、Histogram 通过 recorder helper 注入；
`jiuwenclaw.session.active` 和 `jiuwenclaw.queue.depth` 这类 ObservableGauge
通过 `Observation` 包装逻辑注入。

无身份、身份获取失败、或单个字段为 `None` 时，不输出对应 metric label，
避免产生空字符串或 null 风格的无意义维度。
